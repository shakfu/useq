# Catch2 suites (catch.hpp, single header). Each is one executable and one CTest test.
function(useq_test name source)
    cmake_parse_arguments(ARG "" "TIMEOUT" "" ${ARGN})
    add_executable(${name} ${source})
    target_include_directories(${name} PRIVATE ${PROJECT_SOURCE_DIR} ${PROJECT_SOURCE_DIR}/src)
    target_link_libraries(${name} PRIVATE useq_signal_engine useq_options m)
    add_test(NAME ${name} COMMAND ${name})
    if (ARG_TIMEOUT)
        set_tests_properties(${name} PROPERTIES TIMEOUT ${ARG_TIMEOUT})
    endif ()
endfunction()
